    function generateRandomCode() {
        var length = Math.random() > 0.5 ? 4 : 5;
        var code = '';
        for (var i = 0; i < length; i++) code += Math.floor(Math.random() * 10);
        return code;
    }

    function openResetModal(modalId) {
        var modal = document.getElementById(modalId);
        if (!modal) return;
        modal.style.display = 'flex';
        modal.classList.add('show');
        lockBodyScroll();
        document.getElementById('mainApp').classList.add('blurred');
        pushWindow(modalId);
        history.pushState({ modal: modalId }, '', window.location.href);
    }

    function closeResetModal(modalId) {
        var modal = document.getElementById(modalId);
        if (!modal) return;
        modal.style.display = 'none';
        modal.classList.remove('show');
        windowStack = windowStack.filter(function(w) { return w !== modalId; });
        if (windowStack.length === 0) {
            unlockBodyScroll();
            document.getElementById('mainApp').classList.remove('blurred');
        }
        if (history.state && history.state.modal === modalId) {
            history.replaceState({ modal: windowStack.length > 0 ? windowStack[windowStack.length - 1] : null }, '', window.location.href);
        }
    }

    async function updateDownloadSize() {
        var count = 0,
            totalSize = 0;
        for (var readerId in downloaded) {
            for (var surah in downloaded[readerId]) {
                if (downloaded[readerId][surah] && downloaded[readerId][surah].downloaded) {
                    count++;
                    if (downloaded[readerId][surah].size) totalSize += downloaded[readerId][surah].size;
                }
            }
        }
        if (count > 0 && totalSize === 0) {
            try {
                var cache = await caches.open('quran-audio');
                var keys = await cache.keys();
                for (var request of keys) {
                    var response = await cache.match(request);
                    if (response) {
                        var blob = await response.blob();
                        totalSize += blob.size;
                    }
                }
            } catch (e) {}
        }
        var label = document.getElementById('resetDownloadsLabel');
        if (label) {
            if (count === 0) label.textContent = 'مسح السور المُنزّلة (0 سورة)';
            else if (totalSize > 0) label.textContent = 'مسح السور المُنزّلة (' + count + ' سورة - ' + (
                totalSize / (1024 * 1024)).toFixed(1) + ' م.ب)';
            else label.textContent = 'مسح السور المُنزّلة (' + count + ' سورة - حجم غير معروف)';
        }
    }

    async function getCachedUrl(url) {
        try {
            var cache = await caches.open('quran-audio');
            var cached = await cache.match(url);
            if (cached) return URL.createObjectURL(await cached.blob());
        } catch (e) {}
        return null;
    }

    function getDownloadStats() {
        var readerId = currentReader;
        var count = 0;
        var totalSize = 0;
        if (downloaded[readerId]) {
            for (var idx in downloaded[readerId]) {
                if (downloaded[readerId][idx].downloaded) {
                    count++;
                    if (downloaded[readerId][idx].size) totalSize += downloaded[readerId][idx].size;
                }
            }
        }
        return { count: count, size: totalSize };
    }

    function updateStatsTooltip() {
        var done = 0;
        var total = 0;
        var available = 0;
        var unavailable = 0;
        var failed = 0;
        var unavailableSurahs = [];
        var failedSurahs = [];
        var readerId = currentReader;
        if (surahAvailability[readerId]) {
            for (var i = 0; i < TOTAL_SURAHS; i++) {
                if (surahAvailability[readerId][i] !== undefined) {
                    done++;
                    if (surahAvailability[readerId][i] === 'available' || surahAvailability[readerId][i] === 'cached') available++;
                    else if (surahAvailability[readerId][i] === 'unavailable') {
                        unavailable++;
                        unavailableSurahs.push(i);
                    } else if (surahAvailability[readerId][i] === 'failed') {
                        failed++;
                        failedSurahs.push(i);
                    }
                }
                total++;
            }
        }
        var tooltip = document.getElementById('statsTooltip');
        if (tooltip) {
            var downloadStats = getDownloadStats();
            var sizeText = downloadStats.size > 0 ? formatSize(downloadStats.size) : '0';
            var unavailableHtml = '';
            if (unavailableSurahs.length > 0) {
                var surahList = unavailableSurahs.map(function(idx) {
                    return '<div class="surah-item"><span>' + (idx + 1) + '</span><span>' + surahNames[idx] + '</span></div>';
                }).join('');
                unavailableHtml = '<div class="unavailable-list">❌ غير متوفرة: ' + surahList + '</div>';
            }
            var failedHtml = '';
            if (failedSurahs.length > 0) {
                var failedList = failedSurahs.map(function(idx) {
                    return '<div class="surah-item"><span>' + (idx + 1) + '</span><span>' + surahNames[idx] + '</span></div>';
                }).join('');
                failedHtml = '<div class="unavailable-list" style="border-right-color:#f39c12;color:#f39c12;">⚠️ فشل التحقق: ' + failedList + '</div>';
            }
            tooltip.innerHTML = 
                '<div class="stat-item"><span class="label">📦 متوفر</span><span class="value">' + available + '</span></div>' +
                '<div class="stat-item"><span class="label">❌ غير متوفر</span><span class="value">' + unavailable + '</span></div>' +
                '<div class="stat-item"><span class="label">⚠️ فشل التحقق</span><span class="value">' + failed + '</span></div>' +
                (unavailableHtml) +
                (failedHtml) +
                '<div class="stat-item"><span class="label">⬇️ مُنزّل</span><span class="value">' + downloadStats.count + ' (' + sizeText + ')</span></div>' +
                '<button class="btn-gold-small" id="statsCloseBtn">👍 موافق</button>';
            var closeBtn = document.getElementById('statsCloseBtn');
            if (closeBtn) {

        }

    async function handleSurahAction(e) {
        var target = e.target.closest('button');
        if (!target) return;
        var card = target.closest('.surah-card');
        if (!card) return;
        var idx = parseInt(card.dataset.idx);
        if (isNaN(idx)) return;
        e.stopPropagation();
        if (target.classList.contains('play-btn')) {
            playSurah(idx, 0);
        } else if (target.classList.contains('read-btn')) {
            openReader(idx);
        } else if (target.classList.contains('fav-btn')) {
            toggleFavSurah(idx);
        } else if (target.classList.contains('download-btn')) {
            var isDl = downloaded[currentReader] && downloaded[currentReader][idx] && downloaded[currentReader][idx].downloaded;
            if (isDl) {
                showDeleteConfirm(idx);
                return;
            }
            if (idx < 0 || idx > 113) {
                toastError('رقم سورة غير صحيح');
                return;
            }
            const downloadId = `${currentReader}_${idx}`;
            if (activeDownloads[downloadId]) {
                showAlertHtml(
                    `⏳ التحميل جارٍ بالفعل لسورة <strong>"${surahNames[idx]}"</strong>.<br>هل تريد إلغاء التحميل؟`,
                    function() {
                        const downloadInfo = activeDownloads[downloadId];
                        if (downloadInfo && downloadInfo.xhr) {
                            downloadInfo.xhr.abort();
                            toastInfo('تم إلغاء تحميل "' + surahNames[idx] + '"');
                            if (downloadProgress[currentReader]) {
                                delete downloadProgress[currentReader][idx];
                            }
                            updateSurahCardStatus(idx);
                        }
                    }
                );
                return;
            }
            showDownloadConfirm(idx);
        } else if (target.classList.contains('retry-check-btn')) {
            var idx = parseInt(target.dataset.idx);
            if (!isNaN(idx)) {
                if (!networkOnline) {
                    toastWarning('لا يوجد اتصال بالإنترنت');
                    return;
                }
                if (surahAvailability[currentReader]) {
                    delete surahAvailability[currentReader][idx];
                }
                toastInfo('جاري إعادة التحقق...');
                target.classList.add('checking');
                checkSurahAvailability(currentReader, idx);
                var checkInterval = setInterval(function() {
                    var newStatus = surahAvailability[currentReader] && surahAvailability[
                    currentReader][idx];
                    if (newStatus && newStatus !== 'pending') {
                        clearInterval(checkInterval);
                        target.classList.remove('checking');
                        if (newStatus === 'available' || newStatus === 'cached') {
                            toastSuccess('✅ السورة متوفرة');
                        } else if (newStatus === 'unavailable') {
                            toastWarning('❌ السورة غير موجودة في المصدر');
                        } else if (newStatus === 'failed') {
                            toastError('⚠️ فشل التحقق، تأكد من اتصالك بالإنترنت');
                        }
                    }
                }, 500);
                setTimeout(function() {
                    clearInterval(checkInterval);
                    target.classList.remove('checking');
                    var finalStatus = surahAvailability[currentReader] && surahAvailability[
                        currentReader][idx];
                    if (!finalStatus || finalStatus === 'pending') {
                        toastWarning('تعذر التحقق، تأكد من اتصالك بالإنترنت');
                    }
                }, 15000);
            }
        }
    }

    function playSurah(idx, startTime) {
        startTime = startTime || 0;
        var readerId = currentReader;
        if (idx < 0 || idx > 113) { toastError('رقم سورة غير صحيح'); return; }
        var status = surahAvailability[readerId] && surahAvailability[readerId][idx];
        var url = getReaderUrl(readerId, idx);
        
        caches.open('quran-audio').then(function(cache) {
            cache.match(url).then(function(response) {
                if (response) {
                    playSurahWithRetry(idx, 0, startTime);
                    return;
                }
                if (status === 'unavailable' && repeatMode === 'all') {
                    toastWarning('هذه السورة غير متوفرة لهذا القارئ');
                    var visibleSurahs = [];
                    document.querySelectorAll('.surah-card').forEach(function(card) {
                        if (card.style.display !== 'none') {
                            var i = parseInt(card.dataset.idx);
                            if (!isNaN(i)) visibleSurahs.push(i);
                        }
                    });
                    var currentIndex = visibleSurahs.indexOf(idx);
                    if (currentIndex !== -1 && visibleSurahs.length > 1) {
                        var nextIdx = (currentIndex + 1) % visibleSurahs.length;
                        if (nextIdx !== currentIndex) {
                            setTimeout(function() { playSurah(visibleSurahs[nextIdx], 0); }, 300);
                            return;
                        }
                    }
                    return;
                }
                if (status === 'failed') {
                    toastInfo('فشل التحقق سابقاً، جاري المحاولة...');
                }
                playSurahWithRetry(idx, 0, startTime);
            });
        });
    }

    function toggleFavSurah(idx) {
        var i = favSurahs.indexOf(idx);
        if (i >= 0) {
            favSurahs.splice(i, 1);
            toastSuccess('تم إزالة من المفضلة');
        } else {
            favSurahs.push(idx);
            toastSuccess('تمت الإضافة إلى المفضلة');
        }
        debouncedSaveState();
        renderAllSurahs();
    }

    function startCheckingAvailability() {
        verifyCancelFlag = false;
        var readerId = currentReader;
        if (!surahAvailability[readerId]) surahAvailability[readerId] = {};
        for (var i = 0; i < TOTAL_SURAHS; i++) {
            if (downloaded[readerId] && downloaded[readerId][i] && downloaded[readerId][i].downloaded) {
                surahAvailability[readerId][i] = 'cached';
                updateSurahCardStatus(i);
            } else if (!surahAvailability[readerId][i]) {
                surahAvailability[readerId][i] = 'pending';
                updateSurahCardStatus(i);
            }
        }
        for (var i = 0; i < TOTAL_SURAHS; i++) {
            if (!downloaded[readerId] || !downloaded[readerId][i] || !downloaded[readerId][i].downloaded) {
                checkSurahAvailability(readerId, i);
            }
        }
    }

    function updateVerifyProgress(done, total) {
        const status = document.getElementById('verifyProgressStatus');
        const count = document.getElementById('verifyProgressCount');
        const fill = document.getElementById('verifyProgressFill');
        const retryBtn = document.getElementById('retryFailedBtn');
        const cancelBtn = document.getElementById('cancelVerificationBtn');
        const manualBtn = document.getElementById('manualVerifyBtn');
        if (!status || !count || !fill) return;
        if (!networkOnline) {
            status.textContent = '⚠️ غير متصل - انتظر عودة الإنترنت';
            count.textContent = '';
            fill.style.width = '0%';
            if (cancelBtn) cancelBtn.style.display = 'none';
            if (retryBtn) {
                retryBtn.style.display = 'inline-flex';
                retryBtn.textContent = '🔄 إعادة المحاولة';
                retryBtn.onclick = function() {
                    if (networkOnline) {
                        verifyCancelFlag = false;
                        cancelBtn.style.display = 'inline-flex';
                        retryBtn.style.display = 'none';
                        if (manualBtn) manualBtn.style.display = 'none';
                        startCheckingAvailability();
                    } else {
                        toastWarning('لا يوجد اتصال بالإنترنت');
                    }
                };
            }
            if (manualBtn) manualBtn.style.display = 'none';
            showVerifyProgress();
            return;
        }
        if (done === 0 && total === 0) {
            status.textContent = '⏳ جاري التحقق...';
            count.textContent = '0/114';
            fill.style.width = '0%';
            if (retryBtn) retryBtn.style.display = 'none';
            if (manualBtn) manualBtn.style.display = 'none';
            return;
        }
        if (done >= total) {
            let failedCount = 0;
            const readerId = currentReader;
            if (surahAvailability[readerId]) {
                for (let i = 0; i < TOTAL_SURAHS; i++) {
                    if (surahAvailability[readerId][i] === 'failed') failedCount++;
                }
            }
            if (failedCount > 0) {
                status.textContent = `✅ تم التحقق ⚠️ ${failedCount} فشل`;
                if (retryBtn) {
                    retryBtn.style.display = 'inline-flex';
                    retryBtn.textContent = `🔄 إعادة المحاولة (${failedCount})`;
                    retryBtn.onclick = function() {
                        retryFailedChecks();
                    };
                }
                if (manualBtn) manualBtn.style.display = 'none';
            } else {
                status.innerHTML = '✅ تم التحقق<br><span style="font-size:0.75rem;color:var(--gold);font-weight:600;">👆 اضغط <i data-icon="chart-simple"></i> للإحصائيات</span>';
                if (retryBtn) retryBtn.style.display = 'none';
                if (manualBtn) manualBtn.style.display = 'none';
                const statsBtn = document.getElementById('statsBtn');
                if (statsBtn) {
                    statsBtn.classList.add('pulse');
                    setTimeout(() => {
                        statsBtn.classList.remove('pulse');
                    }, 5000);
                }
                setTimeout(() => {
                    if (document.getElementById('verifyProgressBar')?.style.display !== 'none') {
                        hideVerifyProgress();
                    }
                }, 3000);
            }
            count.textContent = `${done}/${total}`;
            fill.style.width = '100%';
            return;
        }
        status.textContent = `⏳ جاري التحقق... (${Math.round((done/total)*100)}%)`;
        count.textContent = `${done}/${total}`;
        fill.style.width = `${(done/total)*100}%`;
        if (retryBtn) retryBtn.style.display = 'none';
        if (manualBtn) manualBtn.style.display = 'none';
        showVerifyProgress();
    }

    function retryFailedChecks() {
        const readerId = currentReader;
        if (!surahAvailability[readerId]) return;
        if (!networkOnline) {
            toastWarning('لا يوجد اتصال بالإنترنت');
            return;
        }
        let retryCount = 0;
        for (let i = 0; i < TOTAL_SURAHS; i++) {
            if (surahAvailability[readerId][i] === 'failed') {
                delete surahAvailability[readerId][i];
                checkSurahAvailability(readerId, i);
                retryCount++;
            }
        }
        if (retryCount > 0) {
            toastInfo(`جاري إعادة التحقق من ${retryCount} سورة...`);
        } else {
            toastInfo('لا توجد سور فاشلة لإعادة التحقق');
        }
    }
