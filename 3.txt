    async function updateSurahCardStatus(idx) {
        const readerId = currentReader;
        const card = document.querySelector(`.surah-card[data-idx="${idx}"]`);
        if (!card) return;
        const isInCache = await syncSurahWithCache(readerId, idx);
        const statusRow = card.querySelector('.surah-status-row');
        const badge = card.querySelector('.unavailability-badge');
        const warningBadge = card.querySelector('.warning-badge');
        const spinner = card.querySelector('.checking-spinner');
        const downloadBtn = card.querySelector('.download-btn');
        const playBtn = card.querySelector('.play-btn');
        const retryBtn = card.querySelector('.retry-check-btn');
        const status = surahAvailability[readerId]?.[idx];
        badge?.classList.remove('show');
        warningBadge?.classList.remove('show');
        spinner?.classList.remove('show');
        retryBtn?.classList.remove('show');
        downloadBtn?.classList.remove('unavailable', 'verify-failed', 'verify-pending', 'not-exist', 'downloaded');
        playBtn?.classList.remove('disabled', 'verify-pending');
        playBtn.style.opacity = '1';
        playBtn.title = 'تشغيل السورة';
        card.className = 'surah-card';
        const isDl = isInCache;
        const isPartial = downloadProgress[readerId]?.[idx] !== undefined && downloadProgress[readerId][idx] < 100;
        if (isDl) {
            downloadBtn.classList.add('downloaded');
            downloadBtn.innerHTML = '<i data-icon="check-circle"></i>';
            downloadBtn.title = 'حذف التنزيل';
            statusRow.style.display = 'none';
            playBtn.style.opacity = '1';
            playBtn.title = 'تشغيل السورة (محلية)';
            card.classList.add('status-cached');
            applyIconsToElement(downloadBtn);
            return;
        }
        if (isPartial) {
            const pct = Math.round(downloadProgress[readerId][idx]);
            downloadBtn.classList.add('downloading');
            downloadBtn.innerHTML = `
                    <svg class="progress-ring" viewBox="0 0 40 40">
                        <circle class="bg" cx="20" cy="20" r="16"/>
                        <circle class="fg" cx="20" cy="20" r="16" 
                                stroke-dasharray="100.53" 
                                stroke-dashoffset="${100.53 * (1 - pct / 100)}"/>
                    </svg>
                    <span class="progress-text">${pct}%</span>
                    <span class="cancel-icon" data-idx="${idx}">×</span>
                `;
            downloadBtn.title = 'استئناف التنزيل';
            statusRow.style.display = 'none';
            return;
        }
        switch(status) {
            case 'available':
                statusRow.style.display = 'none';
                playBtn.style.opacity = '1';
                playBtn.title = 'تشغيل السورة';
                downloadBtn.innerHTML = '<i data-icon="download"></i>';
                downloadBtn.title = 'تنزيل السورة';
                card.classList.add('status-available');
                break;
            case 'cached':
                statusRow.style.display = 'none';
                playBtn.style.opacity = '1';
                playBtn.title = 'تشغيل السورة (من الذاكرة)';
                downloadBtn.innerHTML = '<i data-icon="download"></i>';
                downloadBtn.title = 'تنزيل السورة';
                card.classList.add('status-cached');
                break;
            case 'unavailable':
                badge.textContent = '❌ غير موجودة في المصدر';
                badge.classList.add('show');
                statusRow.style.display = 'flex';
                playBtn.classList.add('disabled');
                playBtn.style.opacity = '0.3';
                playBtn.title = 'السورة غير متوفرة لهذا القارئ';
                downloadBtn.classList.add('unavailable', 'not-exist');
                downloadBtn.title = 'غير متوفرة للتنزيل';
                card.classList.add('status-unavailable');
                break;
            case 'pending':
                spinner.classList.add('show');
                statusRow.style.display = 'flex';
                playBtn.classList.add('verify-pending');
                playBtn.style.opacity = '0.6';
                playBtn.title = 'جاري التحقق...';
                downloadBtn.classList.add('verify-pending');
                downloadBtn.innerHTML = '<i data-icon="download"></i>';
                downloadBtn.title = 'جاري التحقق...';
                card.classList.add('status-pending');
                break;
            case 'failed':
                warningBadge.textContent = networkOnline ? '⚠️ فشل التحقق (ضعف الاتصال)' : '⚠️ لا يوجد اتصال';
                warningBadge.classList.add('show');
                retryBtn.classList.add('show');
                retryBtn.textContent = networkOnline ? '🔄 إعادة المحاولة' : '📡 انتظر الاتصال';
                statusRow.style.display = 'flex';
                playBtn.style.opacity = '1';
                playBtn.title = 'قد تكون متوفرة، حاول التشغيل';
                downloadBtn.innerHTML = '<i data-icon="download"></i>';
                downloadBtn.title = 'قد تكون متوفرة، حاول التنزيل';
                card.classList.add('status-failed');
                break;
            default:
                statusRow.style.display = 'none';
                playBtn.style.opacity = '1';
                playBtn.title = 'تشغيل السورة';
                downloadBtn.innerHTML = '<i data-icon="download"></i>';
                downloadBtn.title = 'تنزيل السورة';
        }
        if (retryBtn && status === 'failed') {
            retryBtn.onclick = function(e) {
                e.stopPropagation();
                const idx = parseInt(this.dataset.idx);
                if (!isNaN(idx)) {
                    if (!networkOnline) {
                        toastWarning('لا يوجد اتصال بالإنترنت');
                        return;
                    }
                    if (surahAvailability[readerId]) {
                        delete surahAvailability[readerId][idx];
                    }
                    toastInfo('جاري إعادة التحقق...');
                    this.classList.add('checking');
                    checkSurahAvailability(readerId, idx);
                    let checkInterval = setInterval(() => {
                        const newStatus = surahAvailability[readerId]?.[idx];
                        if (newStatus && newStatus !== 'pending') {
                            clearInterval(checkInterval);
                            this.classList.remove('checking');
                            if (newStatus === 'available' || newStatus === 'cached') {
                                toastSuccess('✅ السورة متوفرة');
                            } else if (newStatus === 'unavailable') {
                                toastWarning('❌ السورة غير موجودة في المصدر');
                            } else if (newStatus === 'failed') {
                                toastError('⚠️ فشل التحقق، تأكد من اتصالك بالإنترنت');
                            }
                        }
                    }, 500);
                    setTimeout(() => {
                        clearInterval(checkInterval);
                        this.classList.remove('checking');
                    }, 15000);
                }
            };
        }
        applyIconsToElement(downloadBtn);
        applyIconsToElement(playBtn);
        applyIconsToElement(retryBtn);
    }

    var saveTimeout = null;
    function debouncedSaveState() {
        if (saveTimeout) clearTimeout(saveTimeout);
        saveTimeout = setTimeout(function() {
            saveState();
            saveTimeout = null;
        }, 300);
    }

    function saveState() {
        localStorage.setItem('currentReader', currentReader);
        localStorage.setItem('favSurahs', JSON.stringify(favSurahs));
        localStorage.setItem('favReaders', JSON.stringify(favReaders));
        localStorage.setItem('downloadedSurahs', JSON.stringify(downloaded));
        localStorage.setItem('playSpeed', String(speed));
        localStorage.setItem('playerVolume', String(volume));
        localStorage.setItem('quranFontSize', String(fontsize));
        localStorage.setItem('quranFont', currentFont);
        localStorage.setItem('darkMode', darkMode ? 'dark' : 'light');
        localStorage.setItem('accentColor', currentColor);
        if (currentSurah >= 0 && audio.src && audio.duration) {
            var currentTime = audio.currentTime || 0;
            if (currentTime < audio.duration) {
                localStorage.setItem('lastPlay', JSON.stringify({ 
                    reader: currentReader, 
                    surah: currentSurah, 
                    time: currentTime,
                    timestamp: Date.now()
                }));
            }
        }
    }

    function restoreLast() {
        var last = localStorage.getItem('lastPlay');
        if (last) {
            try {
                var d = JSON.parse(last);
                if (d.reader === currentReader && d.surah >= 0 && d.surah < 114) {
                    lastSurah = d.surah;
                    lastTime = d.time || 0;
                    if (lastTime < 0) lastTime = 0;
                }
            } catch (e) {}
        }
    }

    function saveReadingState(surah, ayah) {
        try {
            localStorage.setItem('lastReading', JSON.stringify({
                surah: surah,
                ayah: ayah,
                timestamp: Date.now()
            }));
        } catch (e) {}
    }

    function restoreReadingState() {
        try {
            var last = localStorage.getItem('lastReading');
            if (last) {
                var d = JSON.parse(last);
                if (d.surah >= 0 && d.surah < 114) {
                    return d;
                }
            }
        } catch (e) {}
        return null;
    }

    function loadFontFamily(font) {
        if (font === 'system') {
            currentFont = 'system';
            document.documentElement.style.setProperty('--font-family', "'Traditional Arabic','Arial',sans-serif");
            document.documentElement.style.setProperty('--font-ui', "'Tajawal','Arial',sans-serif");
            var trigger = document.getElementById('fontSelectTrigger');
            if (trigger) trigger.textContent = 'نظام (تقليدي)';
            document.querySelectorAll('#fontSelectOptions .custom-select-option').forEach(function(opt) {
                opt.classList.toggle('selected', opt.dataset.value === 'system');
            });
            localStorage.setItem('quranFont', 'system');
            return;
        }
        var linkId = 'google-font-' + font.replace(/\s/g, '');
        var link = document.getElementById(linkId);
        if (!link) {
            link = document.createElement('link');
            link.id = linkId;
            link.rel = 'stylesheet';
            link.href = 'https://fonts.googleapis.com/css2?family=' + encodeURIComponent(font) + '&display=swap';
            document.head.appendChild(link);
        }
        currentFont = font;
        document.documentElement.style.setProperty('--font-family', "'" + font + "',serif");
        document.documentElement.style.setProperty('--font-ui', "'" + font + "',sans-serif");
        var trigger = document.getElementById('fontSelectTrigger'),
            options = document.querySelectorAll('#fontSelectOptions .custom-select-option');
        if (trigger) {
            var selectedOption = Array.from(options).find(function(opt) { return opt.dataset.value === font; });
            trigger.textContent = selectedOption ? selectedOption.textContent : font;
        }
        options.forEach(function(opt) { opt.classList.toggle('selected', opt.dataset.value === font); });
        localStorage.setItem('quranFont', font);
    }

    function renderReaderList(filter) {
        var container = $('readerList');
        if (!container) return;
        var q = (filter || '').trim().toLowerCase();
        var entries = readerEntries.slice();
        if (q) entries = entries.filter(function(e) { return e.name.indexOf(q) !== -1; });
        var sorted = entries.sort(function(a, b) {
            var aFav = favReaders.indexOf(a.id) === -1 ? 1 : -1;
            var bFav = favReaders.indexOf(b.id) === -1 ? 1 : -1;
            return (aFav - bFav) || a.name.localeCompare(b.name);
        });
        var frag = document.createDocumentFragment();
        sorted.forEach(function(entry) {
            var div = document.createElement('div');
            div.className = 'reader-item' + (entry.id === currentReader ? ' active' : '');
            var isFav = favReaders.indexOf(entry.id) !== -1;
            var isRadioOn = isRadio && isRadioPlaying && currentReader === entry.id;
            var hasRadio = entry.radio !== null && entry.radio !== undefined && entry.radio.length > 0;
            div.innerHTML =
                '<span class="reader-name">' + entry.name + '</span>' +
                '<span class="fav-star"><i data-icon="star"></i></span>' +
                (hasRadio ?
                    '<span class="reader-radio-btn' + (isRadioOn ? ' playing' : '') + '" data-id="' + entry.id +
                    '" title="' + (isRadioOn ? 'إيقاف الراديو' : 'تشغيل الراديو') + '"><i data-icon="' + (isRadioOn ? 'stop' : 'radio') + '"></i></span>' : 
                    '<span class="reader-radio-btn disabled" title="لا يوجد راديو لهذا القارئ"><i data-icon="radio" style="opacity:0.3;"></i></span>');
            var starEl = div.querySelector('.fav-star');
            if (starEl) {
                starEl.style.opacity = isFav ? '1' : '0.3';
            }
            var radioEl = div.querySelector('.reader-radio-btn');
            if (radioEl && hasRadio) {
                radioEl.addEventListener('click', function(e) {
                    e.stopPropagation();
                    var readerId = this.dataset.id;
                    if (isRadio && isRadioPlaying && currentReader === readerId) {
                        stopRadio();
                    } else {
                        if (isRadio) stopRadio();
                        selectReader(readerId);
                        setTimeout(function() { startRadio(); }, 200);
                    }
                });
            }
            div.addEventListener('click', function() {
                if (!isRadio || !isRadioPlaying || entry.id !== currentReader) {
                    selectReader(entry.id);
                }
                $('readerDropdownMenu').classList.remove('show');
                $('readerDropdownBtn').classList.remove('open');
            });
            frag.appendChild(div);
        });
        container.innerHTML = '';
        container.appendChild(frag);
        applyIconsToAll();
    }

    function getFilterName(filter) {
        var names = {
            'all': 'الكل',
            'available': 'المتوفرة',
            'unavailable': 'غير المتوفرة',
            'downloaded': 'المُنزّلة',
            'notdownloaded': 'غير المُنزّلة'
        };
        return names[filter] || filter;
    }

    function hasFilterResults(filter) {
        var readerId = currentReader;
        var cards = document.querySelectorAll('.surah-card');
        var count = 0;
        cards.forEach(function(card) {
            var idx = parseInt(card.dataset.idx);
            var isDl = downloaded[readerId] && downloaded[readerId][idx] && downloaded[readerId][idx].downloaded;
            var status = surahAvailability[readerId]?.[idx];
            var show = true;
            switch(filter) {
                case 'all': show = true; break;
                case 'available': show = (status === 'available' || status === 'cached'); break;
                case 'unavailable': show = (status === 'unavailable'); break;
                case 'downloaded': show = isDl; break;
                case 'notdownloaded': show = !isDl; break;
            }
            if (show) count++;
        });
        return count > 0;
    }

    var speedMonitor = null,
        currentPlayTimeout = null,
        currentPlayingListener = null;

    function startRadio() {
        var url = getRadioUrl(currentReader);
        if (!url) { toastWarning('لا يوجد راديو لهذا القارئ'); return; }
        if (isRadio) { stopRadio(); return; }
        audio.pause();
        audio.src = '';
        if (speedMonitor) { clearInterval(speedMonitor);
            speedMonitor = null; }
        if (currentPlayTimeout) { clearTimeout(currentPlayTimeout);
            currentPlayTimeout = null; }
        if (currentPlayingListener) { audio.removeEventListener('playing', currentPlayingListener);
            currentPlayingListener = null; }
        isPlaying = false;
        currentSurah = -1;
        updatePlayerUI();
        radioAudio.src = url;
        radioAudio.volume = volume;
        var attempts = 0,
            maxAttempts = 3;
        var tryRadio = function() {
            var playPromise = radioAudio.play();
            if (playPromise !== undefined) {
                playPromise.then(function() {
                    isRadio = true;
                    isRadioPlaying = true;
                    toastSuccess('تم تشغيل راديو ' + getReaderName(currentReader));
                    renderReaderList();
                    updateRadioBtnUI();
                }).catch(function(error) {
                    attempts++;
                    if (attempts < maxAttempts) {
                        toastInfo('جاري إعادة محاولة تشغيل الراديو... (' + attempts + '/' + maxAttempts +
                        ')');
                        setTimeout(tryRadio, 2000);
                    } else {
                        toastError('تعذر تشغيل الراديو. تأكد من سرعة الإنترنت أو جرب لاحقاً.');
                    }
                });
            }
        };
        tryRadio();
    }

    function stopRadio() {

        toast('تم إيقاف الراديو');

    }

    function playSurahWithRetry(idx, attempt, startTime) {
        if (retryLock && attempt > 0) return;
        attempt = attempt || 0;
        startTime = startTime || 0;
        if (isRadio) stopRadio();
        var url = getReaderUrl(currentReader, idx);
        if (!url) { toastError('تعذر الحصول على الرابط'); return; }
        if (currentSurah === idx && audio.src && !audio.paused) {
            audio.pause();
            isPlaying = false;
            updatePlayerUI();
            return;
        }
        if (currentSurah === idx && audio.src && audio.paused) {
            audio.play();
            isPlaying = true;
            updatePlayerUI();
            return;
        }
        if (currentPlayingListener) { audio.removeEventListener('playing', currentPlayingListener);
            currentPlayingListener = null; }
        if (currentPlayTimeout) { clearTimeout(currentPlayTimeout);
            currentPlayTimeout = null; }

        var isCached = false;
        caches.open('quran-audio').then(function(cache) {
            return cache.match(url);
        }).then(function(response) {
            isCached = !!response;
            var tryPlay = function() {
                getCachedUrl(url).then(function(cachedUrl) {
                    audio.src = cachedUrl || url;
                    currentSurah = idx;
                    audio.volume = volume;
                    if (!cachedUrl) audio.load();
                    var timeoutDuration = cachedUrl ? 8000 : 10000;
                    currentPlayTimeout = setTimeout(function() {
                        if (!isPlaying && audio.paused && attempt < 2) {
                            retryLock = true;
                            toastInfo('إعادة محاولة التشغيل...');
                            if (currentPlayingListener) { audio.removeEventListener('playing',
                                    currentPlayingListener);
                                currentPlayingListener = null; }
                            currentPlayTimeout = null;
                            playSurahWithRetry(idx, attempt + 1, startTime);
                        }
                    }, timeoutDuration);

                    function onPlaying() {
                        if (currentPlayTimeout) { clearTimeout(currentPlayTimeout);
                            currentPlayTimeout = null; }
                        retryLock = false;
                        isPlaying = true;
                        updatePlayerUI();
                        if (currentPlayingListener) { audio.removeEventListener('playing',
                                currentPlayingListener);
                            currentPlayingListener = null; }
                        audio.playbackRate = speed;
                        setTimeout(function() { audio.playbackRate = speed; }, 100);
                        if (speedMonitor) clearInterval(speedMonitor);
                        speedMonitor = setInterval(function() {
                            if (audio.playbackRate !== speed && !audio.paused) audio.playbackRate =
                            speed;
                        }, 1000);
                        var infoEl = document.getElementById('playerInfo'),
                            playerEl = document.getElementById('bottomPlayer');
                        if (infoEl) infoEl.textContent = '';
                        if (playerEl) {
                            playerEl.classList.add('visible');
                            updateFooterHeight();
                        }
                        var readerName = getReaderName(currentReader);
                        var surahName = surahNames[idx];
                        if (isCached) {
                            toastSuccess(' تشغيل "' + surahName + '" من الذاكرة 📀');
                        } else {
                            toastSuccess(' تشغيل "' + surahName + '" من الإنترنت 🌐');
                        }
                        addSession('surah', currentReader, idx, startTime);
                        updatePlayerFavButton();
                        updatePlayerTitle();
                    }
                    currentPlayingListener = onPlaying;
                    audio.addEventListener('playing', onPlaying);
                    audio.currentTime = startTime;
                    audio.play().then(function() {}).catch(function(err) {
                        if (currentPlayTimeout) { clearTimeout(currentPlayTimeout);
                            currentPlayTimeout = null; }
                        if (currentPlayingListener) { audio.removeEventListener('playing',
                                currentPlayingListener);
                            currentPlayingListener = null; }
                        if (attempt < 2) {
                            retryLock = true;
                            toastInfo('إعادة محاولة التشغيل... (' + (attempt + 1) + '/2)');
                            setTimeout(function() {
                                retryLock = false;
                                playSurahWithRetry(idx, attempt + 1, startTime);
                            }, 1000 * (attempt + 1));
                        } else {
                            retryLock = false;
                            toastError('فشل التشغيل، تأكد من الاتصال بالإنترنت');
                        }
                    });
                });
            };
            tryPlay();
        }).catch(function() {
            var tryPlay = function() {
                getCachedUrl(url).then(function(cachedUrl) {
                    audio.src = cachedUrl || url;
                    currentSurah = idx;
                    audio.volume = volume;
                    if (!cachedUrl) audio.load();
                    var timeoutDuration = cachedUrl ? 8000 : 10000;
                    currentPlayTimeout = setTimeout(function() {
                        if (!isPlaying && audio.paused && attempt < 2) {
                            retryLock = true;
                            toastInfo('إعادة محاولة التشغيل...');
                            if (currentPlayingListener) { audio.removeEventListener('playing',
                                    currentPlayingListener);
                                currentPlayingListener = null; }
                            currentPlayTimeout = null;
                            playSurahWithRetry(idx, attempt + 1, startTime);
                        }
                    }, timeoutDuration);

                    function onPlaying() {
                        if (currentPlayTimeout) { clearTimeout(currentPlayTimeout);
                            currentPlayTimeout = null; }
                        retryLock = false;
                        isPlaying = true;
                        updatePlayerUI();
                        if (currentPlayingListener) { audio.removeEventListener('playing',
                                currentPlayingListener);
                            currentPlayingListener = null; }
                        audio.playbackRate = speed;
                        setTimeout(function() { audio.playbackRate = speed; }, 100);
                        if (speedMonitor) clearInterval(speedMonitor);
                        speedMonitor = setInterval(function() {
                            if (audio.playbackRate !== speed && !audio.paused) audio.playbackRate =
                            speed;
                        }, 1000);
                        var infoEl = document.getElementById('playerInfo'),
                            playerEl = document.getElementById('bottomPlayer');
                        if (infoEl) infoEl.textContent = '';
                        if (playerEl) {
                            playerEl.classList.add('visible');
                            updateFooterHeight();
                        }
                        var readerName = getReaderName(currentReader);
                        var surahName = surahNames[idx];
                        toastSuccess(' تشغيل "' + surahName + '" من الإنترنت🌐');
                        addSession('surah', currentReader, idx, startTime);
                        updatePlayerFavButton();
                        updatePlayerTitle();
                    }
                    currentPlayingListener = onPlaying;
                    audio.addEventListener('playing', onPlaying);
                    audio.currentTime = startTime;
                    audio.play().then(function() {}).catch(function(err) {
                        if (currentPlayTimeout) { clearTimeout(currentPlayTimeout);
                            currentPlayTimeout = null; }
                        if (currentPlayingListener) { audio.removeEventListener('playing',
                                currentPlayingListener);
                            currentPlayingListener = null; }
                        if (attempt < 2) {
                            retryLock = true;
                            toastInfo('إعادة محاولة التشغيل... (' + (attempt + 1) + '/2)');
                            setTimeout(function() {
                                retryLock = false;
                                playSurahWithRetry(idx, attempt + 1, startTime);
                            }, 1000 * (attempt + 1));
                        } else {
                            retryLock = false;
                            toastError('فشل التشغيل، تأكد من الاتصال بالإنترنت');
                        }
                    });
                });
            };
            tryPlay();
        });
    }

    function resetPlayer(keepExpanded) {
        audio.pause();
        audio.src = '';
        audio.playbackRate = speed;
        if (speedMonitor) { clearInterval(speedMonitor);
            speedMonitor = null; }
        if (currentPlayTimeout) { clearTimeout(currentPlayTimeout);
            currentPlayTimeout = null; }
        if (currentPlayingListener) { audio.removeEventListener('playing', currentPlayingListener);
            currentPlayingListener = null; }
        isPlaying = false;
        currentSurah = -1;
        updatePlayerUI();
        var fill = document.getElementById('playerProgressFill'),
            cur = document.getElementById('playerTimeCur'),
            dur = document.getElementById('playerTimeDur'),
            info = document.getElementById('playerInfo'),
            player = document.getElementById('bottomPlayer');
        if (fill) fill.style.width = '0%';
        if (cur) cur.textContent = '00:00';
        if (dur) dur.textContent = '00:00';
        if (info) info.textContent = '';
        if (player && !keepExpanded) {
            player.classList.remove('visible');
            player.style.bottom = '0px';
            document.body.style.paddingBottom = '5rem';
        }
        updatePlayerFavButton();
        updatePlayerTitle();
    }

    function updateFooterHeight() {
        var footer = document.getElementById('fixedFooter');
        if (!footer) return;
        var footerHeight = footer.offsetHeight;
        document.documentElement.style.setProperty('--footer-height', footerHeight + 'px');
        if (currentSurah >= 0 && !isRadio) {
            var playerEl = document.getElementById('bottomPlayer');
            if (playerEl && playerEl.classList.contains('visible') && !playerEl.classList.contains('modal-mode')) {
                playerEl.style.bottom = footerHeight + 'px';
                document.body.style.paddingBottom = (footerHeight + playerEl.offsetHeight + 32) + 'px';
            }
        }
    }

    function updatePlayerUI() {
        var btn = $('playerPlayBtn');
        if (btn) btn.innerHTML = isPlaying ? '<i data-icon="pause"></i>' : '<i data-icon="play"></i>';
        var infoEl = $('playerInfo');
        if (infoEl) {
            if (currentSurah >= 0 && audio.src) {
                var isCached = audio.src.startsWith('blob:');
                if (isCached) {
                    infoEl.textContent = '📀 من الذاكرة';
                } else {
                    infoEl.textContent = '🌐 من الإنترنت';
                }
            } else {
                infoEl.textContent = '';
            }
        }
        var playerEl = $('bottomPlayer');
        if (playerEl) {
            if (currentSurah >= 0 && !isRadio) {
                playerEl.classList.add('visible');
                updateFooterHeight();
            } else {
                playerEl.classList.remove('visible');
                playerEl.style.bottom = '0px';
                document.body.style.paddingBottom = '5rem';
            }
        }
        updateRadioBtnUI();
        updatePlayerFavButton();
        updatePlayerTitle();
        applyIconsToAll();
    }

            subtitleEl = document.getElementById('playerSubtitle');

    audio.addEventListener('ended', function() {
        if (repeatMode === 'one') {
            audio.currentTime = 0;
            audio.play();
            return;
        }
        if (repeatMode === 'all') {
            var visibleSurahs = [];
            document.querySelectorAll('.surah-card').forEach(function(card) {
                if (card.style.display !== 'none') {
                    var idx = parseInt(card.dataset.idx);
                    if (!isNaN(idx)) visibleSurahs.push(idx);
                }
            });
            if (visibleSurahs.length === 0) {
                isPlaying = false;
                updatePlayerUI();
                return;
            }
            var currentIndex = visibleSurahs.indexOf(currentSurah);
            var nextIndex = (currentIndex + 1) % visibleSurahs.length;
            var nextSurah = visibleSurahs[nextIndex];
            var total = visibleSurahs.length;
            var position = nextIndex + 1;
            toastInfo('⏩ ' + surahNames[nextSurah] + ' (' + position + '/' + total + ')');
            playSurah(nextSurah, 0);
            return;
        }
        isPlaying = false;
        updatePlayerUI();
    });
