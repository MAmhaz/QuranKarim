    function searchInCurrentSurah(query) {
        const container = document.getElementById('readerContent');
        const ayahs = container.querySelectorAll('.reader-ayah');
        clearSearchHighlights();
        readerSearchResults = [];
        readerSearchIndex = -1;
        if (!query || query.length < 2) {
            document.getElementById('readerSearchCount').textContent = '0/0';
            return;
        }
        const normalizedQuery = normalizeText(query);
        var searchTerms = normalizedQuery.split(' ').filter(t => t.length > 0);
        var searchByExact = searchTerms.length > 1;
        ayahs.forEach((ayah, index) => {
            const text = ayah.querySelector('.ayah-text').textContent;
            const normalizedText = normalizeText(text);
            var match = false;
            if (searchByExact) {
                var allTermsMatch = true;
                for (const term of searchTerms) {
                    if (!normalizedText.includes(term)) {
                        allTermsMatch = false;
                        break;
                    }
                }
                if (allTermsMatch) {
                    match = true;
                }
            } else {
                if (normalizedText.includes(searchTerms[0])) {
                    match = true;
                }
            }
            if (match) {
                readerSearchResults.push(index);
                ayah.classList.add('search-highlight');
            }
        });
        if (readerSearchResults.length > 0) {
            readerSearchIndex = 0;
            highlightResult(readerSearchResults[0]);
            document.getElementById('readerSearchCount').textContent = `1/${readerSearchResults.length}`;
        } else {
            document.getElementById('readerSearchCount').textContent = `0/${readerSearchResults.length}`;
        }
    }

    function highlightResult(index) {
        const container = document.getElementById('readerContent');
        const ayahs = container.querySelectorAll('.reader-ayah');
        ayahs.forEach(el => el.classList.remove('search-current'));
        if (index >= 0 && index < ayahs.length) {
            const ayah = ayahs[index];
            ayah.classList.add('search-current');
            ayah.scrollIntoView({ behavior: 'smooth', block: 'center' });
        }
    }

    var mufassirMap = { 'layer1': 'تفسير الميزان (العلامة الطباطبائي)', 'layer2': 'تفسير الميزان (العلامة الطباطبائي)',
        'layer3': 'تفسير معاصر (آية الله مكارم الشيرازي)', 'layer4': 'روايات أهل البيت (الأئمة عليهم السلام)',

    const tafsirCache = new Map();

    async function getTafsirFromFile(surah, ayah) {
        var cacheKey = 'tafsir_' + surah;
        var data = null;
        if (tafsirCache.has(cacheKey)) {
            data = tafsirCache.get(cacheKey);
        } else {
            try {
                var raw = localStorage.getItem('quranTafsir_' + cacheKey);
                if (raw) {
                    var parsed = JSON.parse(raw);
                    if (parsed && parsed.data) {
                        data = parsed.data;
                        tafsirCache.set(cacheKey, data);
                    }
                }
            } catch (e) {}
        }
        if (!data) {
            var url = 'https://raw.githubusercontent.com/MAmhaz/QuranKarim/main/Data/tafsir_' + surah + '.json';
            try {
                var response = await fetch(url);
                if (!response.ok) throw new Error('فشل التحميل');
                data = await response.json();
                tafsirCache.set(cacheKey, data);
                try {
                    localStorage.setItem('quranTafsir_' + cacheKey, JSON.stringify({ data: data, ts: Date
                            .now() }));
                } catch (e) {}
            } catch (e) {
                return { text: 'حدث خطأ في تحميل التفسير', mufassir: '' };
            }
        }
        var ayahData = data[ayah];
        if (!ayahData) return { text: 'الآية غير موجودة', mufassir: '' };
        var key = 'layer1_ar';
        var text = ayahData[key] || 'التفسير غير متوفر';
        var mufassir = mufassirMap['layer1'] || 'غير معروف';
        return { text: text, mufassir: mufassir };
    }

    function generateSurahHtml(data, idx) {
        var ayahs = data.data.ayahs;
        var basmala = (idx === 0) ? '' : (idx === 8 ? '' :
            '<div style="font-family:var(--font-family);font-size:1.3rem;text-align:center;color:var(--gold-dark);margin:.5rem 0;">بِسْمِ اللَّهِ الرَّحْمَٰنِ الرَّحِيمِ</div>'
            );
        var html = basmala;
        ayahs.forEach(function(a) {
            html += '<div class="reader-ayah" data-s="' + (idx + 1) + '" data-a="' + a.numberInSurah +
                '"><span class="ayah-num">' + a.numberInSurah +
                '</span><span class="ayah-text" data-text="' + a.text.replace(/"/g, '&quot;') + '">' + a
                .text + '</span><div class="ayah-actions"></div></div>';
        });
        return html;
    }

    async function loadSurahText(idx) {
        var container = $('readerContent');
        if (!container) return;
        var cacheKey = 'surah_' + idx;
        if (textCache.has(cacheKey)) {
            container.innerHTML = textCache.get(cacheKey);
            attachAyahEvents(container);
            applyIconsToAll();
            return;
        }
        var stored = null;
        try {
            var raw = localStorage.getItem('quranText_' + cacheKey);
            if (raw) {
                var parsed = JSON.parse(raw);
                if (parsed && parsed.html) stored = parsed.html;
            }
        } catch (e) {}
        if (stored) {
            textCache.set(cacheKey, stored);
            container.innerHTML = stored;
            attachAyahEvents(container);
            applyIconsToAll();
            return;
        }
        container.innerHTML = '<div style="display:flex;flex-direction:column;align-items:center;justify-content:center;padding:3rem 0;gap:1rem;"><div class="loading-spinner"></div><span style="color:var(--text-secondary);font-size:0.9rem;">جاري تحميل السورة...</span></div>';
        try {
            var res = await fetch('https://api.alquran.cloud/v1/surah/' + (idx + 1) + '/ar.alafasy');
            var data = await res.json();
            var html = generateSurahHtml(data, idx);
            container.innerHTML = html;
            textCache.set(cacheKey, container.innerHTML);
            try {
                localStorage.setItem('quranText_' + cacheKey, JSON.stringify({ html: container.innerHTML,
                    ts: Date.now() }));
            } catch (e) {}
            attachAyahEvents(container);
            applyIconsToAll();
        } catch (e) {
            container.innerHTML =
                '<div style="text-align:center;padding:2.5rem 0;color:var(--text-secondary);">فشل تحميل السورة</div>';
        }
    }

                    ') —\n\n📚 المفسر: ' + result.mufassir + '\n📝 التفسير: ' + tafsirText +

    function filterSurahs(q) {
        var grid = $('surahGrid');
        if (!grid) return;
        var trimmed = q.trim().toLowerCase();
        if (!trimmed) {
            grid.querySelectorAll('.surah-card').forEach(function(card) { card.style.display = 'flex'; });
            var noResultMsg = document.getElementById('noSurahResult');
            if (noResultMsg) noResultMsg.remove();
            return;
        }
        var parts = trimmed.split(' ').filter(function(p) { return p.length > 0; });
        var cards = grid.querySelectorAll('.surah-card');
        var found = false;
        if (parts.length === 1) {
            var term = parts[0];
            cards.forEach(function(card) {
                var name = card.querySelector('.surah-name').textContent.toLowerCase();
                var match = name.includes(term);
                card.style.display = match ? 'flex' : 'none';
                if (match) found = true;
            });
        } else {
            cards.forEach(function(card) {
                var name = card.querySelector('.surah-name').textContent.toLowerCase();
                var match = parts.every(function(p) { return name.includes(p); });
                card.style.display = match ? 'flex' : 'none';
                if (match) found = true;
            });
        }
        var noResultMsg = document.getElementById('noSurahResult');
        if (!found && trimmed.length > 0) {
            if (!noResultMsg) {
                noResultMsg = document.createElement('div');
                noResultMsg.id = 'noSurahResult';
                noResultMsg.style.cssText = 'text-align:center;padding:2.5rem 0;color:var(--text-secondary);font-size:1rem;';
                noResultMsg.innerHTML = '🔍 لا توجد سور تطابق بحثك<br><span style="font-size:0.8rem;opacity:0.7;">جرب كلمات بحث مختلفة</span>';
                grid.appendChild(noResultMsg);
            }
        } else {
            if (noResultMsg) noResultMsg.remove();
        }
    }

    function updateReaderSearchResults(filter) {
        var resultsContainer = document.getElementById('readerSearchResults');
        if (!resultsContainer) return;

            resultsContainer.classList.remove('show');

        }
        var filtered = readerEntries.filter(function(e) { return e.name.indexOf(q) !== -1; });

            resultsContainer.innerHTML =
                '<div class="reader-search-item" style="color:var(--text-secondary);text-align:center;padding:1.5rem 0;font-size:1rem;">🔍 لا توجد نتائج<br><span style="font-size:0.8rem;opacity:0.7;">جرب كلمات بحث مختلفة</span></div>';

    function getSessions() {
        try { return JSON.parse(localStorage.getItem('sessions') || '[]'); } catch (e) { return []; }
    }

    function saveSessions(sessions) { localStorage.setItem('sessions', JSON.stringify(sessions)); }

    function addSession(type, reader, surah, time) {
        var sessions = getSessions();
        var existing = sessions.findIndex(function(s) {
            return s.type === type && s.reader === reader && (type === 'reader' ? s.surah === surah : true);
        });
        if (existing !== -1) {
            sessions[existing].time = time || 0;
            sessions[existing].surah = surah;
            sessions[existing].timestamp = Date.now();
        } else {
            sessions.push({
                type: type,
                reader: reader || currentReader,
                surah: surah !== undefined ? surah : -1,
                time: time || (audio.currentTime || 0),
                timestamp: Date.now()
            });
        }
        saveSessions(sessions);
    }

    function removeSession(index) {
        var sessions = getSessions();
        sessions.splice(index, 1);
        saveSessions(sessions);
    }

    function checkLastSession() {
        var sessions = getSessions();
        if (sessions.length === 0) return;
        var now = Date.now();
        var validSessions = sessions.filter(function(s) { return (now - s.timestamp) / (1000 * 60 * 60) <= 48; });
        if (validSessions.length === 0) return;
        if (document.querySelector('.resume-overlay')) return;
        var html = '';
        validSessions.forEach(function(session, index) {
            var readerName = getReaderName(session.reader);
            var timeStr = formatTime(session.time || 0);
            if (session.type === 'surah') {
                var surahName = surahNames[session.surah] || 'غير معروف';
                html += '<div class="resume-item"><div><i data-icon="headphones"></i> استماع: سورة ' +
                    surahName + ' (' + readerName +
                    ')</div><div style="display:flex;gap:.5rem;justify-content:center;margin-top:.5rem;flex-wrap:wrap;"><button class="resume-btn" data-index="' +
                    index + '" data-action="play" data-reader="' + session.reader + '" data-surah="' +
                    session.surah + '" data-time="' + session.time + '">▶️ متابعة (' + timeStr +
                    ')</button><button class="resume-btn" data-index="' + index +
                    '" data-action="play" data-reader="' + session.reader + '" data-surah="' + session
                    .surah + '" data-time="0">🔄 إعادة</button><button class="resume-ignore-btn" data-index="' +
                    index + '">⏹️ تجاهل</button></div></div>';
            } else if (session.type === 'reader') {
                var surahName = surahNames[session.surah] || 'غير معروف';
                html += '<div class="resume-item"><div><i data-icon="book-open"></i> قراءة: سورة ' +
                    surahName +
                    '</div><div style="display:flex;gap:.5rem;justify-content:center;margin-top:.5rem;flex-wrap:wrap;"><button class="resume-btn" data-index="' +
                    index + '" data-action="read" data-surah="' + session.surah +
                    '">📖 فتح</button><button class="resume-ignore-btn" data-index="' + index +
                    '">⏹️ تجاهل</button></div></div>';
            }
        });
        if (!html) return;
        var overlay = document.createElement('div');
        overlay.className = 'resume-overlay';
        overlay.style.cssText =
            'position:fixed;inset:0;z-index:999;display:flex;align-items:center;justify-content:center;background:rgba(0,0,0,0.7);backdrop-filter:blur(6px);-webkit-backdrop-filter:blur(6px);padding:1.25rem;';
        overlay.innerHTML =
            '<div style="background:var(--bg-card);border-radius:var(--radius);padding:calc(1.5rem*var(--display-scale));max-width:25rem;width:100%;box-shadow:0 12px 40px rgba(0,0,0,.2);color:var(--text);text-align:center;max-height:80vh;overflow-y:auto;"><h3 style="text-align:center;font-size:calc(1.1rem*var(--text-scale));color:var(--gold-dark);margin-bottom:.75rem;"><i data-icon="history"></i> استئناف الجلسات السابقة</h3><div style="margin:1rem 0;">' +
            html +
            '</div><button class="btn-close-modal" id="sessionClose" style="width:100%;padding:.6rem 2rem;background:var(--gold);border:none;border-radius:2rem;color:#fff;font-family:var(--font-ui);font-weight:700;font-size:calc(.95rem*var(--text-scale));cursor:pointer;transition:all .15s ease;box-shadow:0 2px 12px rgba(201,168,76,.3);margin-top:.5rem;">إغلاق</button></div>';
        document.body.appendChild(overlay);
        document.body.style.overflow = 'hidden';
        document.body.style.position = 'fixed';
        document.body.style.width = '100%';
        document.body.style.height = '100%';
        overlay.querySelectorAll('.resume-btn').forEach(function(btn) {
            btn.addEventListener('click', function() {
                var action = this.dataset.action,
                    index = parseInt(this.dataset.index),
                    readerId = this.dataset.reader,
                    surah = parseInt(this.dataset.surah),
                    time = parseFloat(this.dataset.time) || 0;
                if (action === 'play') {
                    if (readerId) {
                        selectReader(readerId);
                        setTimeout(function() {
                            if (readerId === currentReader && surah === currentSurah)
                                currentSurah = -1;
                            playSurah(surah, time);

    function toggleSideMenu(open) {
        var overlay = $('sideMenuOverlay'),
            menu = $('sideMenu'),
            main = $('mainApp');
        if (!overlay || !menu) return;
        if (open) {
            overlay.classList.add('show');
            menu.classList.add('open');
            if (main) main.classList.add('blurred');
            document.body.style.overflow = 'hidden';
            pushWindow('sideMenu');
            history.pushState({ modal: 'sideMenu' }, '', window.location.href);
        } else {
            overlay.classList.remove('show');
            menu.classList.remove('open');
            if (main) main.classList.remove('blurred');
            document.body.style.overflow = '';
            windowStack = windowStack.filter(function(w) { return w !== 'sideMenu'; });
            if (history.state && history.state.modal === 'sideMenu') {
                history.replaceState({ modal: windowStack.length > 0 ? windowStack[windowStack.length - 1] : null }, '', window.location.href);
            }
        }
    }
