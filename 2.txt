    function hexToRgb(hex) {
        const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
        if (result) {
            const r = parseInt(result[1], 16);
            const g = parseInt(result[2], 16);
            const b = parseInt(result[3], 16);
            return r + ', ' + g + ', ' + b;
        }
        return '201, 168, 76';
    }

    function showToast(message, type, duration) {
        type = type || 'default';
        duration = duration || 2000;
        const t = $('toast');
        if (!t) return;
        t.className = 'toast';
        const icons = { success: '✅', error: '❌', warning: '⚠️', info: 'ℹ️', default: '' };
        const cleanMsg = message.replace(/^[✅❌⚠️ℹ️📦🗑️⏳⏹️🔄▶️⬇️🗑️]+\s*/, '');
        const icon = icons[type] || '';
        t.innerHTML = icon ? '<span>' + icon + ' ' + cleanMsg + '</span>' : '<span>' + cleanMsg + '</span>';
        t.classList.add('show');
        clearTimeout(t._timer);
        if (duration > 0) {
            t._timer = setTimeout(function() {
                t.classList.remove('show');
            }, duration);
        }
    }

    function formatTime(s) {
        if (isNaN(s) || !isFinite(s)) return '00:00';
        var m = Math.floor(s / 60),
            sec = Math.floor(s % 60);
        return String(m).padStart(2, '0') + ':' + String(sec).padStart(2, '0');
    }
    function formatSize(bytes) {
        if (bytes === 0 || !bytes) return 'غير معروف';
        var sizes = ['ب', 'ك.ب', 'م.ب', 'ج.ب'];
        var i = Math.floor(Math.log(bytes) / Math.log(1024));
        return (bytes / Math.pow(1024, i)).toFixed(1) + ' ' + sizes[i];
    }

    function fetchWithTimeout(url, options) {
        options = options || {};
        var timeout = options.timeout || 5000;
        var controller = new AbortController();
        var timeoutId = setTimeout(function() {
            try { controller.abort(); } catch (e) {}
        }, timeout);
        var fetchOptions = {
            method: options.method || 'GET',
            headers: options.headers || {},
            signal: controller.signal
        };
        return fetch(url, fetchOptions).then(function(response) {
            clearTimeout(timeoutId);
            return response;
        }).catch(function(error) {
            clearTimeout(timeoutId);
            throw error;
        });
    }

    function normalizeText(text) {
        return text.replace(/[\u064B-\u065F\u0670]/g, '')
                   .replace(/آ/g, 'ا')
                   .replace(/أ/g, 'ا')
                   .replace(/إ/g, 'ا')
                   .replace(/ٱ/g, 'ا')
                   .replace(/ة/g, 'ه')
                   .replace(/ى/g, 'ي')
                   .replace(/\s+/g, ' ')
                   .trim();
    }

        networkCheckAbort = controller;
        var timeoutId = setTimeout(function() {
            try { controller.abort(); } catch (e) {}
        }, 5000);
        var urls = [
            'https://www.google.com/favicon.ico',
            'https://www.cloudflare.com/favicon.ico',
            'https://1.1.1.1/cdn-cgi/trace'
        ];
        var success = false;
        urls.forEach(function(url, index) {
            if (success) return;
            fetch(url, {
                method: 'HEAD',
                signal: controller.signal,
                cache: 'no-store',
                mode: 'no-cors'
            }).then(function() {
                clearTimeout(timeoutId);
                if (!success) {
                    success = true;
                    if (!networkOnline) {
                        networkOnline = true;
                        updateNetworkIndicator();
                        toastInfo('✅ تم استعادة الاتصال بالإنترنت');
                        if (verifyCancelFlag) {
                            verifyCancelFlag = false;
                            startCheckingAvailability();
                        }
                    }
                    networkCheckAbort = null;
                }
            }).catch(function(error) {
                if (error.name === 'AbortError') {
                    clearTimeout(timeoutId);
                    networkCheckAbort = null;
                    return;
                }
                if (index === urls.length - 1 && !success) {
                    clearTimeout(timeoutId);
                    if (networkOnline) {
                        networkOnline = false;
                        updateNetworkIndicator();
                        toastWarning('⚠️ تم فقدان الاتصال بالإنترنت');
                    }
                    networkCheckAbort = null;
                }
            });
        });
    }

    let iconLoadPromise = null;
    function loadFontAwesome() {
        if (fontAwesomeLoaded) return Promise.resolve();
        if (iconLoadPromise) return iconLoadPromise;
        iconLoadPromise = new Promise(function(resolve, reject) {
            var isLoaded = localStorage.getItem('fontAwesomeLoaded') === 'true';
            var version = localStorage.getItem('fontAwesomeVersion');
            var timestamp = parseInt(localStorage.getItem('fontAwesomeTimestamp') || '0');
            var isExpired = (Date.now() - timestamp) > 7 * 24 * 60 * 60 * 1000;
            if (isLoaded && version === '6.5.0' && !isExpired) {
                fontAwesomeLoaded = true;
                applyIconsToAll();
                resolve();
                return;
            }
            var link = document.createElement('link');
            link.rel = 'stylesheet';
            link.href = 'https://cdnjs.cloudflare.com/ajax/libs/font-awesome/6.5.0/css/all.min.css';
            link.id = 'font-awesome-css';
            document.head.appendChild(link);
            var timeoutId = setTimeout(function() {
                link.onload = null;
                link.onerror = null;
                fontAwesomeLoaded = false;
                applyIconsToAll();
                resolve();
            }, 8000);
            link.onload = function() {
                clearTimeout(timeoutId);
                fontAwesomeLoaded = true;
                try {
                    localStorage.setItem('fontAwesomeLoaded', 'true');
                    localStorage.setItem('fontAwesomeVersion', '6.5.0');
                    localStorage.setItem('fontAwesomeTimestamp', Date.now().toString());
                } catch (e) {}
                applyIconsToAll();
                resolve();
            };
            link.onerror = function() {
                clearTimeout(timeoutId);
                fontAwesomeLoaded = false;
                applyIconsToAll();
                resolve();
            };
        });
        return iconLoadPromise;
    }

    }

    async function syncSurahWithCache(readerId, idx) {
        const url = getReaderUrl(readerId, idx);
        if (!url) return false;
        try {
            const cache = await caches.open('quran-audio');
            const cachedResponse = await cache.match(url);
            const isInCache = !!cachedResponse;
            if (isInCache) {
                let size = 0;
                try {
                    const blob = await cachedResponse.blob();
                    size = blob.size;
                } catch (e) {}
                if (!downloaded[readerId]) downloaded[readerId] = {};
                if (!downloaded[readerId][idx] || !downloaded[readerId][idx].downloaded || downloaded[readerId][idx].size !== size) {
                    downloaded[readerId][idx] = {
                        downloaded: true,
                        size: size || 0,
                        date: new Date().toISOString()
                    };
                    debouncedSaveState();
                }
            } else {
                if (downloaded[readerId] && downloaded[readerId][idx] && downloaded[readerId][idx].downloaded) {
                    delete downloaded[readerId][idx];
                    debouncedSaveState();
                }
            }
            return isInCache;
        } catch (error) {
            return downloaded[readerId]?.[idx]?.downloaded || false;
        }
    }

    async function calculateAllSurahSizes() {
        try {
            const cache = await caches.open('quran-audio');
            const keys = await cache.keys();
            let updated = 0;
            for (const request of keys) {
                const url = request.url;
                for (const reader of readerEntries) {
                    if (url.startsWith(reader.url)) {
                        const idx = parseInt(url.replace(reader.url, '').replace('.mp3', '')) - 1;
                        if (idx >= 0 && idx < 114) {
                            const response = await cache.match(request);
                            if (response) {
                                try {
                                    const blob = await response.blob();
                                    if (blob.size > 0) {
                                        if (!downloaded[reader.id]) downloaded[reader.id] = {};
                                        if (!downloaded[reader.id][idx] || downloaded[reader.id][idx].size !== blob.size) {
                                            downloaded[reader.id][idx] = {
                                                downloaded: true,
                                                size: blob.size,
                                                date: new Date().toISOString()
                                            };
                                            updated++;
                                        }
                                    }
                                } catch (e) {}
                            }
                        }
                        break;
                    }
                }
            }
            if (updated > 0) {
                debouncedSaveState();
            }
        } catch (e) {}
    }

    async function loadQuranTexts() {
        if (quranTexts) return quranTexts;
        const cached = localStorage.getItem('quranTexts');
        if (cached) {
            try {
                quranTexts = JSON.parse(cached);
                return quranTexts;
            } catch (e) {}
        }
        quranTexts = [];
        for (let i = 1; i <= 114; i++) {
            try {
                const response = await fetch(`https://api.alquran.cloud/v1/surah/${i}/ar.alafasy`);
                const data = await response.json();
                quranTexts.push({
                    surah: i,
                    name: surahNames[i-1],
                    ayahs: data.data.ayahs.map((a, idx) => ({
                        number: idx + 1,
                        text: a.text
                    }))
                });
            } catch (e) {}
        }
        try {
            localStorage.setItem('quranTexts', JSON.stringify(quranTexts));
        } catch (e) {}
        return quranTexts;
    }

    async function searchAyahInAllQuran(query) {
        if (!quranTexts) {
            await loadQuranTexts();
        }
        if (!query || query.length < 2) return [];
        const results = [];
        const normalizedQuery = normalizeText(query);
        var searchTerms = normalizedQuery.split(' ').filter(t => t.length > 0);
        var searchByExact = searchTerms.length > 1;
        for (const surah of quranTexts) {
            for (const ayah of surah.ayahs) {
                const normalizedText = normalizeText(ayah.text);
                var match = false;
                if (searchByExact) {
                    var allTermsMatch = true;
                    for (const term of searchTerms) {
                        if (!normalizedText.includes(term)) {
                            allTermsMatch = false;
                            break;
                        }
                    }
                    if (allTermsMatch) {
                        match = true;
                    }
                } else {
                    if (normalizedText.includes(searchTerms[0])) {
                        match = true;
                    }
                }
                if (match) {
                    var score = 0;
                    for (const term of searchTerms) {
                        if (normalizedText.includes(term)) {
                            score++;
                        }
                    }
                    results.push({
                        surah: surah.surah,
                        surahName: surah.name,
                        ayah: ayah.number,
                        text: ayah.text,
                        score: score
                    });
                }
            }
        }
        results.sort((a, b) => b.score - a.score);
        return results;
    }

    function highlightText(text, query) {
        const normalizedQuery = normalizeText(query);
        const terms = normalizedQuery.split(' ').filter(t => t.length > 0);
        let result = text;
        for (const term of terms) {
            const regex = new RegExp(term, 'gi');
            result = result.replace(regex, match => `<span class="highlight-term">${match}</span>`);
        }
        return result;
    }
