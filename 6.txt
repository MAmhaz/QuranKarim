    async function checkSurahAvailability(readerId, idx) {
        if (verifyCancelFlag) return;
        if (!networkOnline) {
            if (!surahAvailability[readerId]) surahAvailability[readerId] = {};
            surahAvailability[readerId][idx] = 'failed';
            updateSurahCardStatus(idx);
            updateGlobalVerifyProgress();
            return;
        }
        if (surahAvailability[readerId] && surahAvailability[readerId][idx] && surahAvailability[readerId][idx] !== 'pending') {
            updateGlobalVerifyProgress();
            return;
        }
        const url = getReaderUrl(readerId, idx);
        if (!url) {
            if (!surahAvailability[readerId]) surahAvailability[readerId] = {};
            surahAvailability[readerId][idx] = 'unavailable';
            updateSurahCardStatus(idx);
            updateGlobalVerifyProgress();
            return;
        }
        if (!surahAvailability[readerId]) surahAvailability[readerId] = {};
        surahAvailability[readerId][idx] = 'pending';
        updateSurahCardStatus(idx);
        updateGlobalVerifyProgress();
        const controller = new AbortController();
        const key = `${readerId}_${idx}`;
        verifyAbortControllers[key] = controller;
        try {
            const cache = await caches.open('quran-audio');
            const cachedResponse = await cache.match(url);
            if (cachedResponse) {
                surahAvailability[readerId][idx] = 'cached';
                if (!downloaded[readerId]) downloaded[readerId] = {};
                downloaded[readerId][idx] = { downloaded: true, size: 0, date: new Date().toISOString() };
                debouncedSaveState();
                updateSurahCardStatus(idx);
                updateGlobalVerifyProgress();
                delete verifyAbortControllers[key];
                return;
            }
            const response = await fetchWithTimeout(url, {
                method: 'HEAD',
                timeout: 5000,
                signal: controller.signal
            });
            if (response.ok) {
                surahAvailability[readerId][idx] = 'available';
                const size = parseInt(response.headers.get('content-length')) || 0;
                if (size > 0) fileSizeCache[url] = size;
            } else if (response.status === 404) {
                surahAvailability[readerId][idx] = 'unavailable';
            } else {
                surahAvailability[readerId][idx] = 'failed';
            }
        } catch (error) {
            if (error.name === 'AbortError') {
                delete surahAvailability[readerId][idx];
            } else {
                surahAvailability[readerId][idx] = 'failed';
            }
        }
        updateSurahCardStatus(idx);
        updateGlobalVerifyProgress();
        delete verifyAbortControllers[key];
    }

    function updateGlobalVerifyProgress() {
        const readerId = currentReader;
        if (!surahAvailability[readerId]) {
            hideVerifyProgress();
            return;
        }
        let total = 0;
        let done = 0;
        let available = 0;
        let unavailable = 0;
        let failed = 0;
        let cached = 0;
        for (let i = 0; i < TOTAL_SURAHS; i++) {
            const status = surahAvailability[readerId][i];
            if (status) {
                total++;
                done++;
                switch(status) {
                    case 'available': available++; break;
                    case 'cached': cached++; break;
                    case 'unavailable': unavailable++; break;
                    case 'failed': failed++; break;
                    case 'pending': done--; break;
                }
            }
        }
        verifyDone = done;
        verifyTotal = total;
        updateVerifyProgress(done, TOTAL_SURAHS);
        const retryBtn = document.getElementById('retryFailedBtn');
        const manualBtn = document.getElementById('manualVerifyBtn');
        if (retryBtn) {
            if (failed > 0 && done >= TOTAL_SURAHS) {
                retryBtn.style.display = 'inline-flex';
                retryBtn.textContent = `🔄 إعادة المحاولة (${failed})`;
                retryBtn.onclick = function() {
                    retryFailedChecks();
                };
            } else if (done === 0 && total === 0 && !verifyCancelFlag) {
                retryBtn.style.display = 'none';
            } else {
                retryBtn.style.display = 'none';
            }
        }
        if (manualBtn) {
            if (verifyCancelFlag || (!networkOnline && done < TOTAL_SURAHS)) {
                manualBtn.style.display = 'inline-flex';
                manualBtn.textContent = '🔍 تحقق';
                manualBtn.onclick = function() {
                    if (!networkOnline) {
                        toastWarning('لا يوجد اتصال بالإنترنت');
                        return;
                    }
                    const readerId2 = currentReader;
                    if (surahAvailability[readerId2]) {
                        for (let i = 0; i < TOTAL_SURAHS; i++) {
                            if (surahAvailability[readerId2][i] === 'failed' || surahAvailability[readerId2][i] === 'unavailable') {
                                delete surahAvailability[readerId2][i];
                            }
                        }
                    }
                    verifyCancelFlag = false;
                    document.getElementById('cancelVerificationBtn').style.display = 'inline-flex';
                    manualBtn.style.display = 'none';
                    document.getElementById('retryFailedBtn').style.display = 'none';
                    startCheckingAvailability();
                    toastInfo('جاري التحقق من جديد...');
                };
            } else {
                manualBtn.style.display = 'none';
            }
        }
        if (done >= TOTAL_SURAHS && failed === 0 && !verifyCancelFlag) {
            setTimeout(() => {
                if (verifyDone >= verifyTotal && !verifyCancelFlag) hideVerifyProgress();
            }, 3000);
        }
    }

    var fileSizeCache = {};

    async function getFileSize(url) {
        if (fileSizeCache[url]) return fileSizeCache[url];
        try {
            var response = await fetchWithTimeout(url, { method: 'HEAD', timeout: 5000 });
            if (response.ok) {
                var size = parseInt(response.headers.get('content-length')) || 0;
                fileSizeCache[url] = size;
                return size;
            }
        } catch (e) {}
        return 0;
    }

    function getFileSizeFromCache(url) { return fileSizeCache[url] || 0; }

    async function startMultipleDownloads(surahIndices) {
        if (isDownloadingMultiple) {
            toastWarning('جاري التنزيل بالفعل');
            return;
        }
        isDownloadingMultiple = true;
        downloadCancel = false;
        var readerId = currentReader;
        var readerName = getReaderName(readerId);
        var overlay = document.getElementById('downloadProgressOverlay');
        document.getElementById('dlProgressReader').textContent = readerName;
        overlay.classList.add('show');
        lockBodyScroll();
        document.getElementById('mainApp').classList.add('blurred');
        pushWindow('downloadProgressOverlay');
        var total = surahIndices.length;
        var done = 0,
            skipped = 0;
        for (var i = 0; i < surahIndices.length; i++) {
            if (downloadCancel) break;
            var idx = surahIndices[i];
            var downloadId = `${readerId}_${idx}`;
            if (activeDownloads[downloadId]) {
                toastWarning(`التحميل جارٍ بالفعل لسورة ${surahNames[idx]}`);
                done++;
                continue;
            }
            var url = getReaderUrl(readerId, idx);
            if (!url) { done++;
                skipped++; continue; }
            var status = surahAvailability[readerId] && surahAvailability[readerId][idx];
            if (status === 'unavailable') { done++;
                skipped++; continue; }
            var isDl = downloaded[readerId] && downloaded[readerId][idx] && downloaded[readerId][idx]
                .downloaded;
            if (isDl) { done++; continue; }
            document.getElementById('dlProgressMain').textContent = Math.round((done / total) * 100) + '%';
            document.getElementById('dlProgressMainBar').style.width = (done / total) * 100 + '%';
            document.getElementById('dlProgressCount').textContent = done + '/' + total;
            document.getElementById('dlCurrentSurah').textContent = surahNames[idx];
            try {
                updateDownloadProgress(readerId, idx, 0);
                var result = await downloadWithProgress(url, function(prog) {
                    updateDownloadProgress(readerId, idx, prog);
                    document.getElementById('dlCurrentSurahBar').style.width = prog + '%';
                    document.getElementById('dlCurrentSurahPct').textContent = prog + '%';
                }, 0, downloadId);
                var blob = result.blob;
                var size = result.size || blob.size;
                var cache = await caches.open('quran-audio');
                await cache.put(url, new Response(blob));
                if (!downloaded[readerId]) downloaded[readerId] = {};
                downloaded[readerId][idx] = { downloaded: true, size: size, date: new Date()
                        .toISOString() };
                clearDownloadProgress(readerId, idx);
                debouncedSaveState();
                done++;
                await updateSurahCardStatus(idx);
                toastSuccess('تم تنزيل: ' + surahNames[idx]);
            } catch (e) {
                if (e.message === 'تم الإلغاء') {
                    toastInfo('تم إيقاف التنزيل');
                    break;
                }
                clearDownloadProgress(readerId, idx);
                delete activeDownloads[downloadId];
                done++;
                skipped++;
                toastError('فشل تنزيل: ' + surahNames[idx]);
            }
        }
        isDownloadingMultiple = false;
        closeModalUniversal('downloadProgressOverlay');
        unlockBodyScroll();
        document.getElementById('mainApp').classList.remove('blurred');
        renderAllSurahs();
        if (!downloadCancel) {
            if (skipped > 0) {
                toastSuccess('تم تنزيل ' + (done - skipped) + ' سورة' + (skipped > 0 ? '، تخطي ' + skipped +
                    ' سورة' : ''));
            } else if (done > 0) {
                toastSuccess('تم تنزيل ' + done + ' سورة بنجاح');
            }
        }
    }

    function downloadWithProgress(url, onProgress, startByte, downloadId) {
        return new Promise(function(resolve, reject) {
            var xhr = new XMLHttpRequest();
            activeDownloads[downloadId] = { xhr: xhr, status: 'downloading' };
            activeXHR = xhr;
            if (startByte && startByte > 0) {
                xhr.open('GET', url, true);
                xhr.setRequestHeader('Range', 'bytes=' + startByte + '-');
            } else {
                xhr.open('GET', url, true);
            }
            xhr.responseType = 'blob';
            var totalSize = 0;
            xhr.onprogress = function(e) {
                if (e.lengthComputable) {
                    totalSize = e.total;
                    onProgress(Math.round((e.loaded / e.total) * 100));
                }
            };
            xhr.onload = function() {
                delete activeDownloads[downloadId];
                activeXHR = null;
                if (xhr.status === 200 || xhr.status === 206) {
                    var blob = xhr.response;
                    resolve({ blob: blob, size: blob.size });
                } else reject(new Error('فشل التنزيل'));
            };
            xhr.onerror = function() {
                delete activeDownloads[downloadId];
                activeXHR = null;
                reject(new Error('خطأ في الشبكة'));
            };
            xhr.onabort = function() {
                delete activeDownloads[downloadId];
                activeXHR = null;
                reject(new Error('تم الإلغاء'));
            };
            xhr.send();
        });
    }

    function updateDownloadProgress(readerId, idx, pct) {
        if (!downloadProgress[readerId]) downloadProgress[readerId] = {};
        downloadProgress[readerId][idx] = pct;
        updateSurahCardStatus(idx);
    }

    function clearDownloadProgress(readerId, idx) {
        if (downloadProgress[readerId]) delete downloadProgress[readerId][idx];
        updateSurahCardStatus(idx);
    }

    function abortCurrentDownload() {
        if (activeXHR) {
            activeXHR.abort();
            activeXHR = null;
        }
    }

    async function cancelPartialDownload(idx) {

        var url = getReaderUrl(readerId, idx);
        if (!url) return;
        abortCurrentDownload();
        var cache = await caches.open('quran-audio');

        if (downloadProgress[readerId]) delete downloadProgress[readerId][idx];
        if (downloaded[readerId]) delete downloaded[readerId][idx];
        debouncedSaveState();
        await updateSurahCardStatus(idx);
        toastSuccess('تم إلغاء التنزيل الجزئي');
    }

    function handleDownloadClick(idx) {
        const readerId = currentReader;
        const downloadId = `${readerId}_${idx}`;
        const surahName = surahNames[idx];
        if (activeDownloads[downloadId]) {
            showAlertHtml(
                `⏳ التحميل جارٍ بالفعل لسورة <strong>"${surahName}"</strong>.<br>هل تريد إلغاء التحميل؟`,
                function() {
                    const downloadInfo = activeDownloads[downloadId];
                    if (downloadInfo && downloadInfo.xhr) {
                        downloadInfo.xhr.abort();
                        toastInfo('تم إلغاء تحميل "' + surahName + '"');
                        if (downloadProgress[readerId]) {
                            delete downloadProgress[readerId][idx];
                        }
                        updateSurahCardStatus(idx);
                    }
                }
            );
            return;
        }
        const url = getReaderUrl(readerId, idx);
        if (!url) {
            toastError('تعذر الحصول على رابط السورة');
            return;
        }
        caches.open('quran-audio').then(function(cache) {
            cache.match(url).then(function(cachedResponse) {
                if (cachedResponse) {
                    if (!downloaded[readerId]) downloaded[readerId] = {};
                    downloaded[readerId][idx] = { downloaded: true, size: 0, date: new Date().toISOString() };
                    debouncedSaveState();
                    updateSurahCardStatus(idx);
                    toastInfo('📀 "' + surahName + '" موجودة بالفعل في الذاكرة.');
                } else {
                    startNewDownload(readerId, idx);
                }
            });
        });
    }

    async function startNewDownload(readerId, idx) {
        const downloadId = `${readerId}_${idx}`;
        if (activeDownloads[downloadId]) {
            toastWarning('التحميل بدأ بالفعل');
            return;
        }
        const url = getReaderUrl(readerId, idx);
        if (!url) {
            toastError('تعذر الحصول على رابط السورة');
            return;
        }
        try {
            const cache = await caches.open('quran-audio');
            const cached = await cache.match(url);
            if (cached) {
                toastInfo('📀 السورة موجودة بالفعل في الذاكرة');
                if (!downloaded[readerId]) downloaded[readerId] = {};
                try {
                    const blob = await cached.blob();
                    downloaded[readerId][idx] = { downloaded: true, size: blob.size, date: new Date().toISOString() };
                } catch (e) {
                    downloaded[readerId][idx] = { downloaded: true, size: 0, date: new Date().toISOString() };
                }
                debouncedSaveState();
                await updateSurahCardStatus(idx);
                return;
            }
        } catch (e) {}
        const surahName = surahNames[idx];
        try {
            updateDownloadProgress(readerId, idx, 0);
            toastInfo('⬇️ جاري تنزيل: ' + surahName);
            var result = await downloadWithProgress(url, function(pct) {
                updateDownloadProgress(readerId, idx, pct);
            }, 0, downloadId);
            var blob = result.blob;
            var size = result.size || blob.size;
            var cache = await caches.open('quran-audio');
            await cache.put(url, new Response(blob));
            if (!downloaded[readerId]) downloaded[readerId] = {};
            downloaded[readerId][idx] = { downloaded: true, size: size, date: new Date().toISOString() };
            clearDownloadProgress(readerId, idx);
            debouncedSaveState();
            await updateSurahCardStatus(idx);
            var sizeText = formatSize(size);
            toastSuccess('✅ تم تنزيل "' + surahName + '" (' + sizeText + ')');
        } catch (e) {
            if (e.message === 'تم الإلغاء') {
                toastInfo('⏹️ تم إلغاء تنزيل: ' + surahName);
            } else {
                toastError('❌ فشل تنزيل: ' + surahName + ' (خطأ: ' + e.message + ')');
            }
            clearDownloadProgress(readerId, idx);
            delete activeDownloads[downloadId];
        }
    }

    function executeDelete() {
        var input = document.getElementById('deleteCaptchaInput').value.trim();
        if (input !== deleteCaptchaCode) {
            document.getElementById('deleteVerifyError').style.display = 'block';
            document.getElementById('deleteCaptchaInput').value = '';
            deleteCaptchaCode = generateRandomCode();
            document.getElementById('deleteCaptchaCode').textContent = deleteCaptchaCode;
            setTimeout(function() { document.getElementById('deleteVerifyError').style.display = 'none'; }, 3000);
            return;
        }
        closeModalUniversal('deleteVerifyModal');
        var checked = document.querySelectorAll('#deleteSelectionGrid input[type="checkbox"]:checked');
        var readerId = currentReader;
        var deleted = 0;
        var totalSize = 0;
        var deletedNames = [];
        checked.forEach(function(cb) {
            var idx = parseInt(cb.dataset.idx);
            var surahName = surahNames[idx];
            var url = getReaderUrl(readerId, idx);
            if (url) {
                caches.open('quran-audio').then(function(cache) {
                    cache.delete(url);
                });
            }
            if (downloaded[readerId] && downloaded[readerId][idx]) {
                totalSize += downloaded[readerId][idx].size || 0;
                delete downloaded[readerId][idx];
                deleted++;
                deletedNames.push(surahName);
            }
            if (surahAvailability[readerId]) {
                delete surahAvailability[readerId][idx];
            }
        });
        if (Object.keys(downloaded[readerId]).length === 0) {
            delete downloaded[readerId];
        }
        debouncedSaveState();
        renderAllSurahs(true);
        var sizeText = totalSize > 0 ? formatSize(totalSize) : '';
        toastSuccess('✅ تم حذف ' + deleted + ' سورة' + (sizeText ? ' (' + sizeText + ')' : ''));
        if (deleted <= 5) {
            toastInfo('السور المحذوفة: ' + deletedNames.join('، '));
        }
    }

    function startSleepTimer(minutes, seconds) {
        stopSleepTimer();
        var totalSeconds = (minutes * 60) + seconds;
        if (totalSeconds < 1) { toastWarning('الحد الأدنى ثانية واحدة'); return; }
        sleepTimerRemaining = totalSeconds;
        document.getElementById('sleepTimerIndicator').style.display = 'flex';
        updateTimerDisplay();
        toastInfo('تم ضبط مؤقت النوم على ' + String(minutes).padStart(2, '0') + ':' + String(seconds).padStart(2,
            '0'));
        sleepTimerInterval = setInterval(function() {
            sleepTimerRemaining--;
            updateTimerDisplay();
            if (sleepTimerRemaining <= 0) {
                stopSleepTimer();
                toastInfo('انتهى وقت المؤقت، تم إيقاف التشغيل');
                if (audio.src && !audio.paused) { audio.pause();

                    updatePlayerUI(); }

                    isRadio = false;
                    isRadioPlaying = false;
                    updateRadioBtnUI(); }
            }
        }, 1000);
    }
